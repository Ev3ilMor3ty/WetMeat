cmake_minimum_required(VERSION 3.16)
project(WetMeatIndependent VERSION 0.2.0 LANGUAGES CXX)
find_package(Threads REQUIRED)
add_executable(wetmeat src/wetmeat.cpp)
target_compile_features(wetmeat PRIVATE cxx_std_17)
target_link_libraries(wetmeat PRIVATE Threads::Threads)
if(MSVC)
  target_compile_options(wetmeat PRIVATE /W4 /O2)
else()
  target_compile_options(wetmeat PRIVATE -Wall -Wextra -Wpedantic)
endif()
