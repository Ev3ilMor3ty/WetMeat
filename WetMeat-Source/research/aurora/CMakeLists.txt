cmake_minimum_required(VERSION 3.16)
project(Aurora VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(AURORA_NATIVE "Optimize for the build machine's CPU" OFF)
find_package(Threads REQUIRED)
add_library(aurora_core src/chess.cpp src/evaluate.cpp src/search.cpp)
target_include_directories(aurora_core PUBLIC src)
target_link_libraries(aurora_core PUBLIC Threads::Threads)
if(MSVC)
    target_compile_options(aurora_core PRIVATE /W4)
else()
    target_compile_options(aurora_core PRIVATE -Wall -Wextra -Wpedantic)
    if(AURORA_NATIVE)
        target_compile_options(aurora_core PRIVATE -march=native)
    endif()
endif()
add_executable(aurora src/main.cpp)
target_link_libraries(aurora PRIVATE aurora_core)
include(CTest)
if(BUILD_TESTING)
    add_executable(core-tests tests/core_tests.cpp)
    target_link_libraries(core-tests PRIVATE aurora_core)
    add_test(NAME chess-rules COMMAND core-tests --extended)
    find_package(Python3 COMPONENTS Interpreter QUIET)
    if(Python3_FOUND)
        add_test(NAME uci-protocol COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_uci.py --engine $<TARGET_FILE:aurora>)
    endif()
endif()
install(TARGETS aurora RUNTIME DESTINATION bin)
